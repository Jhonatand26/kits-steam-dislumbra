% Placa de circuito con agujeros en las esquinas.
% #1 x0, #2 y0, #3 x1, #4 y1, #5 color
\newcommand{\pcb}[5]{%
  \draw[pieza, fill=#5, rounded corners=1.5pt] (#1,#2) rectangle (#3,#4);
  \fill[white] (#1+0.14,#2+0.14) circle (0.06);
  \fill[white] (#3-0.14,#2+0.14) circle (0.06);
  \fill[white] (#1+0.14,#4-0.14) circle (0.06);
  \fill[white] (#3-0.14,#4-0.14) circle (0.06);%
}
% Fila de pines hacia abajo. #1 x inicial, #2 y, #3 cantidad
\newcommand{\pines}[3]{%
  \foreach \i in {1,...,#3}{%
    \draw[line width=0.9pt, draw=oro] (#1+\i*0.28-0.28,#2)
      -- ++(0,-0.45);}%
  \fill[pcbnegro] (#1-0.1,#2-0.12) rectangle (#1+#3*0.28-0.18,#2+0.04);%
}
